		\multicolumn{2}{|l|}{\textbf{Tên Use Case: Đăng Ký}} \\

		Đăng ký tài khoản \\
		\hline
		
		\textbf{Tác nhân:} &
		Bé (6 -- 10 tuổi) \\
		\hline
		
		\textbf{Điều kiện trước:} &
		Truy cập vào trang đăng ký của hệ thống thành công \\

		Đã mở ứng dụng thành công \\

			2. Hệ thống hiển thị form đăng ký.
			
			3. Người dùng nhập tên bé, tuổi, email phụ huynh và mật khẩu.
			
			4. Người dùng nhấn nút Đăng ký.
			
			5. Hệ thống kiểm tra dữ liệu nhập vào.
			
			6. Nếu dữ liệu hợp lệ, hệ thống lưu tài khoản vào SQLite.
			
			7. Hệ thống thông báo đăng ký thành công.
			
			8. Hệ thống chuyển sang màn hình đăng nhập.
		\end{minipage}
		\\
		\hline
		
		\textbf{Luồng sự kiện phụ:} &
		\begin{minipage}[t]{\linewidth}
			4.1. Nếu bỏ trống thông tin, hệ thống báo không được bỏ trống.
			
			4.2. Nếu tuổi không nằm trong khoảng 6 đến 10, hệ thống báo tuổi không hợp lệ.
			
			4.3. Nếu email sai định dạng hệ thống sẽ báo lỗi.
			
			4.4. Nếu mật khẩu dưới 6 ký tự hoặc thiếu chữ số, ký tự đặc biệt, số, hệ thống sẽ báo lỗi.
			
			5.1. Nếu email tồn tại, hệ thống báo tài khoản đã được đăng ký.
		\end{minipage}
		\\
		\hline
